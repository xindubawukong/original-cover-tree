cmake_minimum_required(VERSION 3.16)
project(original_cover_tree LANGUAGES CXX)
add_library(original_cover_tree STATIC cover_tree.cc)
target_include_directories(original_cover_tree PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_features(original_cover_tree PUBLIC cxx_std_11)
if(CMAKE_SOURCE_DIR STREQUAL PROJECT_SOURCE_DIR)
  add_executable(knn knn.cc point.cc)
  target_link_libraries(knn PRIVATE original_cover_tree)
  add_executable(test_nn test_nn.cc point.cc)
  target_link_libraries(test_nn PRIVATE original_cover_tree)
  include(CTest)
  if(BUILD_TESTING)
    find_package(Threads REQUIRED)
    add_executable(cover_tree_test tests/cover_tree_test.cc)
    target_link_libraries(cover_tree_test PRIVATE original_cover_tree Threads::Threads)
    add_test(NAME cover_tree_test COMMAND cover_tree_test)
  endif()
endif()
